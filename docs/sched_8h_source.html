69 template <class Rep, class Period, class Func>
70 void schedule(
71 const std::chrono::duration<Rep, Period> &duration, Func &&func) {
72 std::thread(
73 [this](std::chrono::duration<Rep, Period> duration, Func func) {
74 auto deadline = std::chrono::steady_clock::now() + duration;
75 std::unique_lock<std::mutex> lock(mut);
76 bool interrupted = cv.wait_until(lock, deadline, [this] {
77 return canceled.load(std::memory_order_acquire);
78 });
79 if (!interrupted) {
80 lock.unlock();
81 func();
82 }
83 },
84 duration,
85 std::forward<Func>(func))
86 .detach();
87 }

123 template <class Rep, class Period, class Func>
124 void interval(
125 const std::chrono::duration<Rep, Period> &duration, Func &&func) {
126 mut.lock();
127 threads.push_back(std::thread(
128 [this](std::chrono::duration<Rep, Period> duration, Func func) {
129 std::unique_lock<std::mutex> lock(mut, std::adopt_lock);
130 auto deadline = std::chrono::steady_clock::now();
131 while (true) {
132 deadline += duration;
133 bool interrupted = cv.wait_until(
134 lock, deadline, [this] {
135 return canceled.load(std::memory_order_acquire);
136 });
137 if (interrupted) break;
138 lock.unlock();
139 func();
140 lock.lock();
141 }
142 },
143 duration,
144 std::forward<Func>(func)));
145 }
