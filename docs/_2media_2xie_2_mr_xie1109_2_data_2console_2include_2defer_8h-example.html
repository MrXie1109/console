#pragma once
#include <type_traits>
#include <utility>
namespace console {
template <class F>
class DeferGuard {
F func;
public:
DeferGuard(const F &f) : func(f) {}
DeferGuard(F &&f) : func(std::move(f)) {}
~DeferGuard() { func(); }
};
template <class F>
DeferGuard<typename std::decay<F>::type> make_defer(F &&f) {
return std::forward<F>(f);
}
