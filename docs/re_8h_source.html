83 * @class Match
84 * @brief 匹配结果对象，保存文本副本和位置信息，不依赖迭代器有效性。
85 */
86 class Match {
87 private:
88 string_type text_copy;
89 struct GroupInfo {
90 std::ptrdiff_t pos;
91 std::ptrdiff_t len;
92 };
93 std::vector<GroupInfo> groups_;
94 bool success;
95
96 public:
100 Match() : success(false) {}
101
104 * @param t 文本副本。
105 * @param m 匹配结果。
106 */
107 Match(string_type t, const match_type &m) :
108 text_copy(std::move(t)), success(true) {
109 for (size_t i = 0; i < m.size(); ++i)
110 groups_.push_back({m.position(i), m.length(i)});
111 }
112
115 * @param n 组的索引，默认为 0。
116 * @return 匹配成功时返回组的子串，否则返回空字符串。
117 */
118 string_type group(int n = 0) const {
119 if (!success || n < 0 || n >= int(groups_.size()))
120 return string_type{};
121 return text_copy.substr(groups_[n].pos, groups_[n].len);
122 }
123
125 * @brief 获取所有组的子串。
126 * @return 匹配成功时返回所有组的子串列表，否则返回空列表。
127 */
128 std::vector<string_type> groups() const {
129 std::vector<string_type> result;
130 if (!success) return result;
131 for (size_t i = 1; i < groups_.size(); ++i)
132 result.push_back(
133 text_copy.substr(groups_[i].pos, groups_[i].len));
134 return result;
135 }
136
139 * @param n 组的索引，默认为 0。
140 * @return 匹配成功时返回组的起始位置，否则返回 -1。
141 */
142 int start(int n = 0) const {
143 if (!success || n < 0 || n >= int(groups_.size())) return -1;
144 return int(groups_[n].pos);
145 }
146
149 * @param n 组的索引，默认为 0。
150 * @return 匹配成功时返回组的结束位置，否则返回 -1。
151 */
152 int end(int n = 0) const {
153 if (!success || n < 0 || n >= int(groups_.size())) return -1;
154 return int(groups_[n].pos + groups_[n].len);
155 }
156
159 * @param n 组的索引，默认为 0。
160 * @return 匹配成功时返回组的起始和结束位置，否则返回 (-1, -1)。
161 */
162 std::pair<int, int> span(int n = 0) const {
163 return {start(n), end(n)};
164 }
165
169
170 explicit operator bool() const { return success; }
171 };

208 * @param text 要搜索的文本。
209 * @return 匹配成功时返回匹配对象列表，否则返回空列表。
210 */
211 std::vector<string_type> findall(const string_type &text) const {
212 std::vector<string_type> result;
213 auto begin = iterator_type(text.begin(), text.end(), pattern);
214 auto end = iterator_type();
215 for (auto it = begin; it != end; ++it) result.push_back(it->str());
216 return result;
217 }
218
223 * @return 分割后的字符串列表。
224 */
225 std::vector<string_type>
226 split(const string_type &text, int maxsplit = 0) const {
227 std::vector<string_type> result;
228 token_iterator it(text.begin(), text.end(), pattern, -1);
229 token_iterator end;
230 int count = 0;
231 for (; it != end && (maxsplit <= 0 || count < maxsplit);
232 ++it, ++count)
233 result.push_back(*it);
234 if (maxsplit > 0 && count == maxsplit && it != end)
235 result.push_back(string_type((*it).first, text.end()));
236 return result;
237 }
238
243 * @param count 替换次数，默认为 0 表示替换所有匹配。
244 * @return 替换后的字符串。
245 */
246 string_type sub(const string_type &repl,
247 const string_type &text,
248 int count = 0) const {
249 if (count <= 0) return std::regex_replace(text, pattern, repl);
250 string_type result;
251 auto begin = iterator_type(text.begin(), text.end(), pattern);
252 auto end = iterator_type();
253 auto last = text.begin();
254 int replaced = 0;
255 for (auto it = begin; it != end && replaced < count;
256 ++it, ++replaced) {
257 result.append(last, text.begin() + it->position());
258 result.append(repl);
259 last = text.begin() + it->position() + it->length();
260 }
261 result.append(last, text.end());
262 return result;
263 }
264
269 * @param count 替换次数，默认为 0 表示替换所有匹配。
270 * @return 替换后的字符串和替换次数的对。
271 */
272 std::pair<string_type, int> subn(const string_type &repl,
273 const string_type &text,
274 int count = 0) const {
275 if (count <= 0) {
276 auto result = std::regex_replace(text, pattern, repl);
277 auto begin = iterator_type(text.begin(), text.end(), pattern);
278 auto end = iterator_type();
279 int cnt = int(std::distance(begin, end));
280 return {result, cnt};
281 }
282 string_type result;
283 auto begin = iterator_type(text.begin(), text.end(), pattern);
284 auto end = iterator_type();
285 auto last = text.begin();
286 int replaced = 0;
287 for (auto it = begin; it != end && replaced < count;
288 ++it, ++replaced) {
289 result.append(last, text.begin() + it->position());
290 result.append(repl);
291 last = text.begin() + it->position() + it->length();
292 }
293 result.append(last, text.end());
294 return {result, replaced};
295 }
