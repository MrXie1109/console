256 inline double fps(double target) {
257 using namespace std::chrono;
258 thread_local auto last = steady_clock::now(); //于首次调用时初始化
259 std::this_thread::sleep_until(last + duration<double>(1.0 / target));
260 double actual
261 = 1.0 / duration<double>(steady_clock::now() - last).count();
262 last = steady_clock::now();
263 return actual;

368 int iso_week() const {
369 // 取当年 1 月 4 日所在的周一为准
370 DateTime jan4(year(), 1, 4);
371 int dow = jan4.iso_weekday();
372 int week_num = (yearday() - jan4.yearday() + dow + 6) / 7;
373 if (week_num < 1)
374 week_num = DateTime(year() - 1, 12, 31).iso_week();
375 if (week_num > 52 && days_in_month() == 31 && month() == 12
376 && iso_weekday() <= 3)
377 week_num = 1;
378 return week_num;
379 }
