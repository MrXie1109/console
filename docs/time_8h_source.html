259 inline double fps(double target) {
260 using namespace std::chrono;
261 thread_local auto last = steady_clock::now(); //于首次调用时初始化
262 std::this_thread::sleep_until(last + duration<double>(1.0 / target));
263 double actual
264 = 1.0 / duration<double>(steady_clock::now() - last).count();
265 last = steady_clock::now();
266 return actual;

371 int iso_week() const {
372 // 取当年 1 月 4 日所在的周一为准
373 DateTime jan4(year(), 1, 4);
374 int dow = jan4.iso_weekday();
375 int week_num = (yearday() - jan4.yearday() + dow + 6) / 7;
376 if (week_num < 1)
377 week_num = DateTime(year() - 1, 12, 31).iso_week();
378 if (week_num > 52 && days_in_month() == 31 && month() == 12
379 && iso_weekday() <= 3)
380 week_num = 1;
381 return week_num;
382 }
