100 inline bool miller_rabin_test(uint64_t n, uint64_t a) noexcept {
101 if (n < 2) return false;
102 if (n == 2 || n == 3) return true;
103 if (n % 2 == 0) return false;
104 uint64_t d = n - 1;
105 int s = 0;
106 while (d % 2 == 0) {
107 d /= 2;
108 s++;
109 }
110 uint64_t x = mod_pow(a, d, n);
111 if (x == 1 || x == n - 1) return true;
112 for (int r = 1; r < s; r++) {
113 x = mod_mul(x, x, n);
114 if (x == n - 1) return true;
115 if (x == 1) return false;
116 }
117 return false;
118 }
