55 struct PairHash {
63 template <typename T1, typename T2>
64 std::size_t operator()(std::pair<T1, T2> p) const {
65 auto h1 = std::hash<T1>{}(p.first);
66 auto h2 = std::hash<T2>{}(p.second);
67 return h1 ^ (h2 << 1);
68 }
69 };

171#else

173#endif
