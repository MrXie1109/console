59 inline std::string tiname_impl(const std::type_info &ti) {
60#ifdef _MSC_VER
61 return ti.name();
62#elif defined(__GNUG__)
63 int status = 0;
64 std::unique_ptr<char, void (*)(void *)> result(
65 abi::__cxa_demangle(ti.name(), nullptr, nullptr, &status),
66 std::free);
67 return (status == 0) ? result.get() : ti.name();
68#else
69 return ti.name();
70#endif
71 }
