179 template <class Gen>
180 class Window : public Generator<Window<Gen>,
181 std::vector<typename Gen::value_type>> {
182 Gen gen_;
183 size_t window_size_;
184 std::vector<typename Gen::value_type> window_;
185 bool started = false;
186
187 void init() {
188 if (!started) {
189 for (size_t i = 0; i < window_size_ && !gen_.done(); ++i) {
190 window_.push_back(gen_.current());
191 gen_.advance();
192 }
193 started = true;
194 }
195 }
196
197 public:
204 explicit Window(Gen gen, size_t window_size = 16) :
205 gen_(gen), window_size_(window_size) {
206 if (window_size_ == 0)
207 throw ValueError("Window size must be greater than 0");
208 }
209
214 bool done() {
215 init();
216 return window_.size() < window_size_;
217 }
218
223 std::vector<typename Gen::value_type> current() {
224 init();
225 return window_;
226 }
227
231 void advance() {
232 init();
233 if (!gen_.done()) {
234 window_.erase(window_.begin());
235 window_.push_back(gen_.current());
236 gen_.advance();
237 } else
238 window_.erase(window_.begin());
239 }
240 };
