63 template <class T, size_t... Dims>
64 double mean(const MultiArray<T, Dims...> &arr) {
65 return double(sum(arr)) / arr.fsize();

77 template <class T, size_t... Dims>
78 double variance(const MultiArray<T, Dims...> &arr, bool sample = true) {
79 double m = mean(arr);
80 double sq_sum = 0;
81 arr.for_each([&](const T &x) {
82 double d = double(x) - m;
83 sq_sum += d * d;
84 });
85 return sq_sum / (arr.fsize() - (sample ? 1 : 0));
86 }

97 template <class T, size_t... Dims>
98 double stddev(const MultiArray<T, Dims...> &arr, bool sample = true) {
99 return std::sqrt(variance(arr, sample));

151 double len = norm(a);
152 if (std::abs(len) < std::numeric_limits<double>::epsilon()) return a;
153 MultiArray<T, N> result;

155 auto ri = result.fbegin();

726 template <class T, size_t... Dims>
727 double covariance(
728 const MultiArray<T, Dims...> &a, const MultiArray<T, Dims...> &b) {
729 double mean_a = mean(a);
730 double mean_b = mean(b);
731 double sum = 0;
732 auto ai = a.fbegin();
733 auto bi = b.fbegin();
734 while (ai != a.fend()) {
735 sum += (double(*ai) - mean_a) * (double(*bi) - mean_b);
736 ++ai;
737 ++bi;
738 }
739 return sum / (a.fsize() - 1);
740 }

750 template <class T, size_t... Dims>
751 double correlation(
752 const MultiArray<T, Dims...> &a, const MultiArray<T, Dims...> &b) {
753 double cov = covariance(a, b);
754 double std_a = stddev(a);
755 double std_b = stddev(b);
756 if (std_a == 0 || std_b == 0) return 0;
757 return cov / (std_a * std_b);
758 }

769 size_t n = arr.fsize();

785 enum class QuantileMethod {
786 Nearest,
787 Linear,
788 Midpoint,
789 Lower,
790 Upper,
791 ExcelInc,
792 ExcelExc
793 };
794
804 template <class T, size_t... Dims>
805 double quantile(MultiArray<T, Dims...> arr,
806 double q,
807 QuantileMethod method = QuantileMethod::Nearest) {
808 size_t n = arr.fsize();
809 if (n == 0) return 0;
810 if (q < 0) q = 0;
811 if (q > 1) q = 1;
812 auto begin = arr.fbegin();
813 auto end = arr.fend();
814 switch (method) {
815 case QuantileMethod::Nearest: {
816 size_t idx = size_t(q * (n - 1) + 0.5);
817 if (idx >= n) idx = n - 1;
818 std::nth_element(begin, begin + idx, end);
819 return double(arr.fbegin()[idx]);
820 }
821 case QuantileMethod::Linear: {
822 double pos = q * (n - 1);
823 size_t idx = size_t(pos);
824 double frac = pos - idx;
825 if (idx >= n - 1) {
826 std::nth_element(begin, begin + n - 1, end);
827 return double(arr.fbegin()[n - 1]);
828 }
829 std::nth_element(begin, begin + idx + 1, end);
830 double a = double(arr.fbegin()[idx]);
831 double b = double(arr.fbegin()[idx + 1]);
832 return a + (b - a) * frac;
833 }
834 case QuantileMethod::Midpoint: {
835 double pos = q * (n - 1);
836 size_t idx = size_t(pos);
837 if (idx >= n - 1) {
838 std::nth_element(begin, begin + n - 1, end);
839 return double(arr.fbegin()[n - 1]);
840 }
841 std::nth_element(begin, begin + idx + 1, end);
842 double a = double(arr.fbegin()[idx]);
843 double b = double(arr.fbegin()[idx + 1]);
844 return (a + b) / 2;
845 }
846 case QuantileMethod::Lower: {
847 size_t idx = size_t(q * (n - 1));
848 if (idx >= n) idx = n - 1;
849 std::nth_element(begin, begin + idx, end);
850 return double(arr.fbegin()[idx]);
851 }
852 case QuantileMethod::Upper: {
853 size_t idx = size_t(q * (n - 1));
854 if (idx >= n - 1) return double(arr.fbegin()[n - 1]);
855 idx++;
856 std::nth_element(begin, begin + idx, end);
857 return double(arr.fbegin()[idx]);
858 }
859 case QuantileMethod::ExcelInc: {
860 double pos = (n - 1) * q;
861 size_t idx = size_t(pos);
862 double frac = pos - idx;
863 if (idx >= n - 1) {
864 return double(arr.fbegin()[n - 1]);
865 }
866 std::nth_element(begin, begin + idx + 1, end);
867 double a = double(arr.fbegin()[idx]);
868 double b = double(arr.fbegin()[idx + 1]);
869 return a + (b - a) * frac;
870 }
871 case QuantileMethod::ExcelExc: {
872 double pos = (n + 1) * q;
873 size_t idx = size_t(pos);
874 double frac = pos - idx;
875 if (idx < 1) idx = 1;
876 if (idx >= n) {
877 if (idx == n) {
878 std::nth_element(begin, begin + n - 1, end);
879 return double(arr.fbegin()[n - 1]);
880 }
881 return double(arr.fbegin()[n - 1]);
882 }
883 std::nth_element(begin, begin + idx, end);
884 double a = double(arr.fbegin()[idx - 1]);
885 double b = double(arr.fbegin()[idx]);
886 return a + (b - a) * frac;
887 }
888 default:
889 return quantile(arr, q, QuantileMethod::Nearest);
890 }

937 template <class T>
938 MultiArray<T, 3, 3> inv3x3(const MultiArray<T, 3, 3> &A) {
939 T d = det3x3(A);
940 if (d == 0) return MultiArray<T, 3, 3>{};
941 MultiArray<T, 3, 3> B;
942 B[0][0] = (A[1][1] * A[2][2] - A[1][2] * A[2][1]) / d;
943 B[0][1] = (A[0][2] * A[2][1] - A[0][1] * A[2][2]) / d;
944 B[0][2] = (A[0][1] * A[1][2] - A[0][2] * A[1][1]) / d;
945 B[1][0] = (A[1][2] * A[2][0] - A[1][0] * A[2][2]) / d;
946 B[1][1] = (A[0][0] * A[2][2] - A[0][2] * A[2][0]) / d;
947 B[1][2] = (A[0][2] * A[1][0] - A[0][0] * A[1][2]) / d;
948 B[2][0] = (A[1][0] * A[2][1] - A[1][1] * A[2][0]) / d;
949 B[2][1] = (A[0][1] * A[2][0] - A[0][0] * A[2][1]) / d;
950 B[2][2] = (A[0][0] * A[1][1] - A[0][1] * A[1][0]) / d;
951 return B;
952 }

998 template <class T, size_t... Dims>
999 MultiArray<T, Dims...> softmax(const MultiArray<T, Dims...> &arr) {
1000 auto max_val = max(arr);
1001 MultiArray<T, Dims...> exp_arr;
1002 auto ri = exp_arr.fbegin();
1003 auto ai = arr.fbegin();
1004 while (ri != exp_arr.fend()) {
1005 *ri++ = std::exp(*ai++ - max_val);

1078 template <class T, size_t N>
1079 MultiArray<std::complex<T>, N>
1080 fft(const MultiArray<std::complex<T>, N> &x) {
1081 static_assert((N & (N - 1)) == 0, "FFT length must be a power of 2");
1082 MultiArray<std::complex<T>, N> result = x;
1083 for (size_t i = 0; i < N; ++i) {
1084 size_t j = 0;
1085 for (size_t k = 1; k < N; k <<= 1) {
1086 if (i & k)
1087 j = (j << 1) | 1;
1088 else
1089 j <<= 1;
1090 }
1091 if (i < j) std::swap(result[i], result[j]);
1092 }
1093 for (size_t len = 2; len <= N; len <<= 1) {
1094 T angle = T(-2 * M_PI) / len;
1095 std::complex<T> wlen(std::cos(angle), std::sin(angle));
1096 for (size_t i = 0; i < N; i += len) {
1097 std::complex<T> w = 1;
1098 for (size_t j = 0; j < len / 2; ++j) {
1099 auto u = result[i + j];
1100 auto v = result[i + j + len / 2] * w;
1101 result[i + j] = u + v;
1102 result[i + j + len / 2] = u - v;
1103 w *= wlen;
1104 }
1105 }
1106 }
1107 return result;
1108 }
