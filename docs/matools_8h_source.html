63 template <class T, size_t... Dims>
64 double mean(const MultiArray<T, Dims...> &arr) {
65 return double(sum(arr)) / arr.fsize();
66 }

77 template <class T, size_t... Dims>
78 double variance(const MultiArray<T, Dims...> &arr, bool sample = true) {
79 double m = mean(arr);
80 double sq_sum = 0;
81 arr.for_each([&](const T &x) {
82 double d = double(x) - m;
83 sq_sum += d * d;
84 });
85 return sq_sum / (arr.fsize() - (sample ? 1 : 0));
86 }

97 template <class T, size_t... Dims>
98 double stddev(const MultiArray<T, Dims...> &arr, bool sample = true) {
99 return std::sqrt(variance(arr, sample));
100 }

170 double sum = 0;

721 template <class T, size_t... Dims>
722 double covariance(
723 const MultiArray<T, Dims...> &a, const MultiArray<T, Dims...> &b) {
724 double mean_a = mean(a);
725 double mean_b = mean(b);
726 double sum = 0;
727 auto ai = a.fbegin();
728 auto bi = b.fbegin();
729 while (ai != a.fend()) {
730 sum += (double(*ai) - mean_a) * (double(*bi) - mean_b);
731 ++ai;
732 ++bi;
733 }
734 return sum / (a.fsize() - 1);
735 }

745 template <class T, size_t... Dims>
746 double correlation(
747 const MultiArray<T, Dims...> &a, const MultiArray<T, Dims...> &b) {
748 double cov = covariance(a, b);
749 double std_a = stddev(a);
750 double std_b = stddev(b);
751 if (std_a == 0 || std_b == 0) return 0;
752 return cov / (std_a * std_b);
753 }

764 size_t n = arr.fsize();
765 auto begin = arr.fbegin();
766 auto end = arr.fend();

780 enum class QuantileMethod {
781 Nearest,
782 Linear,
783 Midpoint,
784 Lower,
785 Upper,
786 ExcelInc,
787 ExcelExc
788 };
789
799 template <class T, size_t... Dims>
800 double quantile(MultiArray<T, Dims...> arr,
801 double q,
802 QuantileMethod method = QuantileMethod::Nearest) {
803 size_t n = arr.fsize();
804 if (n == 0) return 0;
805 if (q < 0) q = 0;
806 if (q > 1) q = 1;
807 auto begin = arr.fbegin();
808 auto end = arr.fend();
809 switch (method) {
810 case QuantileMethod::Nearest: {
811 size_t idx = size_t(q * (n - 1) + 0.5);
812 if (idx >= n) idx = n - 1;
813 std::nth_element(begin, begin + idx, end);
814 return double(arr.fbegin()[idx]);
815 }
816 case QuantileMethod::Linear: {
817 double pos = q * (n - 1);
818 size_t idx = size_t(pos);
819 double frac = pos - idx;
820 if (idx >= n - 1) {
821 std::nth_element(begin, begin + n - 1, end);
822 return double(arr.fbegin()[n - 1]);
823 }
824 std::nth_element(begin, begin + idx + 1, end);
825 double a = double(arr.fbegin()[idx]);
826 double b = double(arr.fbegin()[idx + 1]);
827 return a + (b - a) * frac;
828 }
829 case QuantileMethod::Midpoint: {
830 double pos = q * (n - 1);
831 size_t idx = size_t(pos);
832 if (idx >= n - 1) {
833 std::nth_element(begin, begin + n - 1, end);
834 return double(arr.fbegin()[n - 1]);
835 }
836 std::nth_element(begin, begin + idx + 1, end);
837 double a = double(arr.fbegin()[idx]);
838 double b = double(arr.fbegin()[idx + 1]);
839 return (a + b) / 2;
840 }
841 case QuantileMethod::Lower: {
842 size_t idx = size_t(q * (n - 1));
843 if (idx >= n) idx = n - 1;
844 std::nth_element(begin, begin + idx, end);
845 return double(arr.fbegin()[idx]);
846 }
847 case QuantileMethod::Upper: {
848 size_t idx = size_t(q * (n - 1));
849 if (idx >= n - 1) return double(arr.fbegin()[n - 1]);
850 idx++;
851 std::nth_element(begin, begin + idx, end);
852 return double(arr.fbegin()[idx]);
853 }
854 case QuantileMethod::ExcelInc: {
855 double pos = (n - 1) * q;
856 size_t idx = size_t(pos);
857 double frac = pos - idx;
858 if (idx >= n - 1) {
859 return double(arr.fbegin()[n - 1]);
860 }
861 std::nth_element(begin, begin + idx + 1, end);
862 double a = double(arr.fbegin()[idx]);
863 double b = double(arr.fbegin()[idx + 1]);
864 return a + (b - a) * frac;
865 }
866 case QuantileMethod::ExcelExc: {
867 double pos = (n + 1) * q;
868 size_t idx = size_t(pos);
869 double frac = pos - idx;
870 if (idx < 1) idx = 1;
871 if (idx >= n) {
872 if (idx == n) {
873 std::nth_element(begin, begin + n - 1, end);
874 return double(arr.fbegin()[n - 1]);
875 }
876 return double(arr.fbegin()[n - 1]);
877 }
878 std::nth_element(begin, begin + idx, end);
879 double a = double(arr.fbegin()[idx - 1]);
880 double b = double(arr.fbegin()[idx]);
881 return a + (b - a) * frac;
882 }
883 default:
884 return quantile(arr, q, QuantileMethod::Nearest);
885 }
886 }

932 template <class T>
933 MultiArray<T, 3, 3> inv3x3(const MultiArray<T, 3, 3> &A) {
934 T d = det3x3(A);
935 if (d == 0) return MultiArray<T, 3, 3>{};
936 MultiArray<T, 3, 3> B;
937 B[0][0] = (A[1][1] * A[2][2] - A[1][2] * A[2][1]) / d;
938 B[0][1] = (A[0][2] * A[2][1] - A[0][1] * A[2][2]) / d;
939 B[0][2] = (A[0][1] * A[1][2] - A[0][2] * A[1][1]) / d;
940 B[1][0] = (A[1][2] * A[2][0] - A[1][0] * A[2][2]) / d;
941 B[1][1] = (A[0][0] * A[2][2] - A[0][2] * A[2][0]) / d;
942 B[1][2] = (A[0][2] * A[1][0] - A[0][0] * A[1][2]) / d;
943 B[2][0] = (A[1][0] * A[2][1] - A[1][1] * A[2][0]) / d;
944 B[2][1] = (A[0][1] * A[2][0] - A[0][0] * A[2][1]) / d;
945 B[2][2] = (A[0][0] * A[1][1] - A[0][1] * A[1][0]) / d;
946 return B;
947 }

993 template <class T, size_t... Dims>
994 MultiArray<T, Dims...> softmax(const MultiArray<T, Dims...> &arr) {
995 auto max_val = max(arr);
996 MultiArray<T, Dims...> exp_arr;
997 auto ri = exp_arr.fbegin();
998 auto ai = arr.fbegin();
999 while (ri != exp_arr.fend()) {
1000 *ri++ = std::exp(*ai++ - max_val);
1001 }
1002 T sum_exp = sum(exp_arr);
1003 MultiArray<T, Dims...> result;

1073 template <class T, size_t N>
1074 MultiArray<std::complex<T>, N>
1075 fft(const MultiArray<std::complex<T>, N> &x) {
1076 static_assert((N & (N - 1)) == 0, "FFT length must be a power of 2");
1077 MultiArray<std::complex<T>, N> result = x;
1078 for (size_t i = 0; i < N; ++i) {
1079 size_t j = 0;
1080 for (size_t k = 1; k < N; k <<= 1) {
1081 if (i & k)
1082 j = (j << 1) | 1;
1083 else
1084 j <<= 1;
1085 }
1086 if (i < j) std::swap(result[i], result[j]);
1087 }
1088 for (size_t len = 2; len <= N; len <<= 1) {
1089 T angle = T(-2 * M_PI) / len;
1090 std::complex<T> wlen(std::cos(angle), std::sin(angle));
1091 for (size_t i = 0; i < N; i += len) {
1092 std::complex<T> w = 1;
1093 for (size_t j = 0; j < len / 2; ++j) {
1094 auto u = result[i + j];
1095 auto v = result[i + j + len / 2] * w;
1096 result[i + j] = u + v;
1097 result[i + j + len / 2] = u - v;
1098 w *= wlen;
1099 }
1100 }
1101 }
1102 return result;
1103 }
