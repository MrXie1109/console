79 template <class T, size_t... Dims>
80 double variance(const MultiArray<T, Dims...> &arr, bool sample = true) {
81 double m = mean(arr);
82 double sq_sum = 0;
83 arr.for_each([&](const T &x) {
84 double d = double(x) - m;
85 sq_sum += d * d;
86 });
87 return sq_sum / (arr.fsize() - (sample ? 1 : 0));
88 }

213 template <class T, size_t M, size_t N, size_t K>
214 MultiArray<T, M, K>
215 matmul_blocked(const MultiArray<T, M, N> &A, const MultiArray<T, N, K> &B) {
216 using std::min;
217 MultiArray<T, M, K> C(T{});
218 constexpr size_t BLOCK = CONSOLE_MATMUL_BLOCK_SIZE;
219 for (size_t i0 = 0; i0 < M; i0 += BLOCK)
220 for (size_t j0 = 0; j0 < K; j0 += BLOCK)
221 for (size_t k0 = 0; k0 < N; k0 += BLOCK)
222 for (size_t i = i0; i < min(i0 + BLOCK, M); ++i)
223 for (size_t j = j0; j < min(j0 + BLOCK, K); ++j)
224 for (size_t k = k0; k < min(k0 + BLOCK, N); ++k)
225 C[i][j] += A[i][k] * B[k][j];
226 return C;
227 }

313 template <class T, size_t M, size_t N, size_t K>
314 MultiArray<T, M, K> matmul_parallel(
315 const MultiArray<T, M, N> &A, const MultiArray<T, N, K> &B) {
316 using std::min;
317 MultiArray<T, M, K> C(T{});
318 constexpr size_t BLOCK = CONSOLE_MATMUL_BLOCK_SIZE;
319 const size_t num_i_blocks = (M + BLOCK - 1) / BLOCK;
320 const size_t num_j_blocks = (K + BLOCK - 1) / BLOCK;
321 std::vector<std::future<void>> futures;
322 futures.reserve(num_i_blocks * num_j_blocks);
323 for (size_t i0_idx = 0; i0_idx < num_i_blocks; ++i0_idx)
324 for (size_t j0_idx = 0; j0_idx < num_j_blocks; ++j0_idx) {
325 size_t i0 = i0_idx * BLOCK;
326 size_t j0 = j0_idx * BLOCK;
327 futures.push_back(pool::submit([&, i0, j0]() {
328 for (size_t k0 = 0; k0 < N; k0 += BLOCK)
329 for (size_t i = i0; i < min(i0 + BLOCK, M); ++i)
330 for (size_t j = j0; j < min(j0 + BLOCK, K); ++j) {
331 T sum = C[i][j];
332 for (size_t k = k0; k < min(k0 + BLOCK, N); ++k)
333 sum += A[i][k] * B[k][j];
334 C[i][j] = sum;
335 }
336 }));
337 }
338 for (auto &f : futures) f.get();
339 return C;
340 }

846 template <class T, size_t... Dims>
847 double covariance(
848 const MultiArray<T, Dims...> &a, const MultiArray<T, Dims...> &b) {
849 double mean_a = mean(a);
850 double mean_b = mean(b);
851 double sum = 0;
852 auto ai = a.fbegin();
853 auto bi = b.fbegin();
854 while (ai != a.fend()) {
855 sum += (double(*ai) - mean_a) * (double(*bi) - mean_b);
856 ++ai;
857 ++bi;
858 }
859 return sum / (a.fsize() - 1);
860 }

870 template <class T, size_t... Dims>
871 double correlation(
872 const MultiArray<T, Dims...> &a, const MultiArray<T, Dims...> &b) {
873 double cov = covariance(a, b);
874 double std_a = stddev(a);
875 double std_b = stddev(b);
876 if (std_a == 0 || std_b == 0) return 0;
877 return cov / (std_a * std_b);
878 }

905 enum class QuantileMethod {
906 Nearest,
907 Linear,
908 Midpoint,
909 Lower,
910 Upper,
911 ExcelInc,
912 ExcelExc
913 };
914
924 template <class T, size_t... Dims>
925 double quantile(MultiArray<T, Dims...> arr,
926 double q,
927 QuantileMethod method = QuantileMethod::Nearest) {
928 size_t n = arr.fsize();
929 if (n == 0) return 0;
930 if (q < 0) q = 0;
931 if (q > 1) q = 1;
932 auto begin = arr.fbegin();
933 auto end = arr.fend();
934 switch (method) {
935 case QuantileMethod::Nearest: {
936 size_t idx = size_t(q * (n - 1) + 0.5);
937 if (idx >= n) idx = n - 1;
938 std::nth_element(begin, begin + idx, end);
939 return double(arr.fbegin()[idx]);
940 }
941 case QuantileMethod::Linear: {
942 double pos = q * (n - 1);
943 size_t idx = size_t(pos);
944 double frac = pos - idx;
945 if (idx >= n - 1) {
946 std::nth_element(begin, begin + n - 1, end);
947 return double(arr.fbegin()[n - 1]);
948 }
949 std::nth_element(begin, begin + idx + 1, end);
950 double a = double(arr.fbegin()[idx]);
951 double b = double(arr.fbegin()[idx + 1]);
952 return a + (b - a) * frac;
953 }
954 case QuantileMethod::Midpoint: {
955 double pos = q * (n - 1);
956 size_t idx = size_t(pos);
957 if (idx >= n - 1) {
958 std::nth_element(begin, begin + n - 1, end);
959 return double(arr.fbegin()[n - 1]);
960 }
961 std::nth_element(begin, begin + idx + 1, end);
962 double a = double(arr.fbegin()[idx]);
963 double b = double(arr.fbegin()[idx + 1]);
964 return (a + b) / 2;
965 }
966 case QuantileMethod::Lower: {
967 size_t idx = size_t(q * (n - 1));
968 if (idx >= n) idx = n - 1;
969 std::nth_element(begin, begin + idx, end);
970 return double(arr.fbegin()[idx]);
971 }
972 case QuantileMethod::Upper: {
973 size_t idx = size_t(q * (n - 1));
974 if (idx >= n - 1) return double(arr.fbegin()[n - 1]);
975 idx++;
976 std::nth_element(begin, begin + idx, end);
977 return double(arr.fbegin()[idx]);
978 }
979 case QuantileMethod::ExcelInc: {
980 double pos = (n - 1) * q;
981 size_t idx = size_t(pos);
982 double frac = pos - idx;
983 if (idx >= n - 1) {
984 return double(arr.fbegin()[n - 1]);
985 }
986 std::nth_element(begin, begin + idx + 1, end);
987 double a = double(arr.fbegin()[idx]);
988 double b = double(arr.fbegin()[idx + 1]);
989 return a + (b - a) * frac;
990 }
991 case QuantileMethod::ExcelExc: {
992 double pos = (n + 1) * q;
993 size_t idx = size_t(pos);
994 double frac = pos - idx;
995 if (idx < 1) idx = 1;
996 if (idx >= n) {
997 if (idx == n) {
998 std::nth_element(begin, begin + n - 1, end);
999 return double(arr.fbegin()[n - 1]);
1000 }
1001 return double(arr.fbegin()[n - 1]);
1002 }
1003 std::nth_element(begin, begin + idx, end);
1004 double a = double(arr.fbegin()[idx - 1]);
1005 double b = double(arr.fbegin()[idx]);
1006 return a + (b - a) * frac;
1007 }
1008 default:
1009 return quantile(arr, q, QuantileMethod::Nearest);
1010 }

1198 template <class T, size_t N>
1199 MultiArray<std::complex<T>, N>
1200 fft(const MultiArray<std::complex<T>, N> &x) {
1201 static_assert((N & (N - 1)) == 0, "FFT length must be a power of 2");
1202 MultiArray<std::complex<T>, N> result = x;
1203 for (size_t i = 0; i < N; ++i) {
1204 size_t j = 0;
1205 for (size_t k = 1; k < N; k <<= 1) {
1206 if (i & k)
1207 j = (j << 1) | 1;
1208 else
1209 j <<= 1;
1210 }
1211 if (i < j) std::swap(result[i], result[j]);
1212 }
1213 for (size_t len = 2; len <= N; len <<= 1) {
1214 T angle = T(-2 * M_PI) / len;
1215 std::complex<T> wlen(std::cos(angle), std::sin(angle));
1216 for (size_t i = 0; i < N; i += len) {
1217 std::complex<T> w = 1;
1218 for (size_t j = 0; j < len / 2; ++j) {
1219 auto u = result[i + j];
1220 auto v = result[i + j + len / 2] * w;
1221 result[i + j] = u + v;
1222 result[i + j + len / 2] = u - v;
1223 w *= wlen;
1224 }
1225 }
1226 }
1227 return result;
1228 }
