160 string_type read_text() const {
161 std::ifstream fin(narrow());
162 if (!fin.is_open())
163 throw FileError("Cannot Open File \"" + narrow() + '"');
164 std::string buf{std::istreambuf_iterator<char>(fin),
165 std::istreambuf_iterator<char>()};
166 if (fin.fail() && !fin.eof())
167 throw FileError(
168 "The Stream of \"" + narrow() + "\" is Not Good");
169 return widen_or_pass(buf);
170 }
171
176 Bytes read_binary() const {

178 if (!fin.is_open())
179 throw FileError("Cannot Open File \"" + narrow() + '"');

181 std::istreambuf_iterator<char>()};

183 throw FileError(
184 "The Stream of \"" + narrow() + "\" is Not Good");
185 return bytes;
186 }

202 template <class T>
203 T read_POD() const {
204 static_assert(std::is_trivially_copyable<T>::value,
205 "This Type is Not POD Type!");
206 std::ifstream fin(narrow(), std::ios::binary);
207 if (!fin.is_open())
208 throw FileError("Cannot Open File \"" + narrow() + '"');
209 T data;

266 void write_lines(const std::vector<string_type> &lines) const {
267 std::ofstream fout(narrow(), std::ios::binary);
268 if (lines.empty()) return;
269 if (!fout.is_open())
270 throw FileError("Cannot Open File \"" + narrow() + '"');
271 write_to_stream(fout, lines[0]);
272 for (size_t i = 1; i < lines.size(); ++i) {
273 fout << '\n';
274 write_to_stream(fout, lines[i]);
275 }
276 if (fout.fail())
277 throw FileError(
278 "The Stream of \"" + narrow() + "\" is Not Good");
