93 inline std::basic_ostream<CharT, Traits> &cont_print_sequence(
94 std::basic_ostream<CharT, Traits> &os, const Cont &cont) {
95 if (begin(cont) == end(cont)) return os << CharT('[') << CharT(']');

98 repr(*it, os);
