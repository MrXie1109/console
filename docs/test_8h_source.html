70 inline void pre_summary() {
71 struct Guard {
72 Guard() { std::cout << "=== Start Test ===\n" << std::endl; }
73
74 ~Guard() {
75 int passed = passed_count();
76 int failed = failed_count();
77 int total = passed + failed;
78 std::cout << "=== Test Summary ===" << std::endl;
79 std::cout << "Passed: " << passed << std::endl;
80 std::cout << "Failed: " << failed << std::endl;
81 std::cout << "Total: " << total << std::endl;
82 if (total > 0)
83 std::cout
84 << "Rate: " << std::fixed << std::setprecision(1)
85 << (static_cast<double>(passed) / total * 100) << "%"
86 << std::endl;
87 else
88 std::cout << "Rate: N/A" << std::endl;
89 }
90 };
91 static Guard guard;
92 }
