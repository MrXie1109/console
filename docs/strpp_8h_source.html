100 std::basic_string<CharT, Traits, Allocator>
101 trim(std::basic_string<CharT, Traits, Allocator> str) {
102 auto front = std::find_if(str.begin(),
103 str.end(),
104 [](unsigned char uc) -> bool { return !isspace(uc); });
105 if (front == str.end()) {
106 str.clear();
107 return str;
108 }
109 auto back = std::find_if(str.rbegin(),
110 str.rend(),
111 [](unsigned char uc) -> bool { return !isspace(uc); });
112 str.erase(back.base(), str.end());
113 str.erase(str.begin(), front);
114 return str;

236 template <class CharT, class Traits, class Allocator>
237 std::basic_string<CharT, Traits, Allocator>
238 title(std::basic_string<CharT, Traits, Allocator> str,
239 const std::locale &loc = std::locale{}) {
240 using string_type = std::basic_string<CharT, Traits, Allocator>;
241 if (str.empty()) return string_type{};
242 str[0] = std::toupper(str[0], loc);
243 for (size_t i = 1; i < str.size(); ++i) {
244 if (std::isspace(str[i - 1], loc))
245 str[i] = std::toupper(str[i], loc);
246 else
247 str[i] = std::tolower(str[i], loc);
248 }
249 return str;
250 }

298 template <class CharT, class Traits, class Allocator>
299 BasicPartitionResult<CharT, Traits, Allocator>
300 partition(const std::basic_string<CharT, Traits, Allocator> &text,
301 const std::basic_string<CharT, Traits, Allocator> &sep) {
302 using string_type = std::basic_string<CharT, Traits, Allocator>;
303 size_t pos = text.find(sep);
304 if (pos == string_type::npos)
305 return BasicPartitionResult<CharT, Traits, Allocator>{
306 text, string_type{}, string_type{}};
307 return BasicPartitionResult<CharT, Traits, Allocator>{
308 text.substr(0, pos), sep, text.substr(pos + sep.size())};
309 }

321 template <class CharT, class Traits, class Allocator>
322 std::vector<std::basic_string<CharT, Traits, Allocator>>
323 split(const std::basic_string<CharT, Traits, Allocator> &text,
324 const std::basic_string<CharT, Traits, Allocator> &sep
325 = std::basic_string<CharT, Traits, Allocator>(1, CharT{' '})) {
326 using string_type = std::basic_string<CharT, Traits, Allocator>;
327 using vector_type = std::vector<string_type>;
328 vector_type result;
329 if (sep.empty()) return {text};
330 size_t start = 0;
331 while (start <= text.size()) {
332 size_t pos = text.find(sep, start);
333 if (pos == string_type::npos) {
334 result.push_back(text.substr(start));
335 break;
336 }
337 result.push_back(text.substr(start, pos - start));
338 start = pos + sep.size();
339 }
340 return result;
341 }
