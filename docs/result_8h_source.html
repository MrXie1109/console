323 void swap(Result &other) noexcept {
324 if (this == &other) return;
325 if (is_ok_ && other.is_ok_) {
326 T temp = std::move(*get_ok_ptr());
327 *get_ok_ptr() = std::move(*other.get_ok_ptr());
328 *other.get_ok_ptr() = std::move(temp);
329 } else if (!is_ok_ && !other.is_ok_) {
330 E temp = std::move(*get_err_ptr());
331 *get_err_ptr() = std::move(*other.get_err_ptr());
332 *other.get_err_ptr() = std::move(temp);
333 } else if (is_ok_) {
334 T temp_t = std::move(*get_ok_ptr());
335 get_ok_ptr()->~T();
336 new (get_err_ptr()) E(std::move(*other.get_err_ptr()));
337 other.get_err_ptr()->~E();
338 new (other.get_ok_ptr()) T(std::move(temp_t));
339 } else {
340 E temp_e = std::move(*get_err_ptr());
341 get_err_ptr()->~E();
342 new (get_ok_ptr()) T(std::move(*other.get_ok_ptr()));
343 other.get_ok_ptr()->~T();
344 new (other.get_err_ptr()) E(std::move(temp_e));
345 }
346 std::swap(is_ok_, other.is_ok_);
347 }
